## 1. 已有框架接口总结

### 1.1 核心接口与扩展点

ScholarSprout 框架定义了 **5 个核心扩展点**，论文精读模块需要对接其中 4 个：

| 扩展点 | 接口定义 | 已有示例 | 论文精读需要做什么 |
|--------|----------|----------|-------------------|
| **Agent Profile** | `agents/profiles.json` 中的 JSON 条目 | `chat`, `domain_onboarding` | 新增 `paper_reading` agent profile |
| **Handler** | `(ChannelMessage, app_state) -> dict` | `handle_chat_message`, `handle_domain_onboarding_message` | 重写 `handle_paper_reading_message` |
| **Skill (SKILL.md)** | `skills/builtin/{category}/{name}/SKILL.md` | `chat.default`, `chat.research_discussion` | 创建 8 个 reading 类别 skill |
| **Tool** | `BaseTool` + `ToolSpec` + `run(arguments)` | `CurrentTimeTool` | 创建论文搜索、PDF 解析、KG 查询等 tool |
| **API Route** | `gateway/app.py` 中的 FastAPI 路由 | `POST /chat`, `POST /domain_onboarding` | 扩展 `POST /paper_reading`，新增论文管理路由 |

### 1.2 消息流（已有，无需修改）

```
HTTP Request → route → process_channel_input()
  → channel.receive_message() → ChannelMessage(inbound)
  → run_in_threadpool(process_channel_message)
    → channel.publish_inbound() → bus 记录
    → handler(message, app_state) → dict
    → build_channel_output() → ChannelMessage(outbound)
    → channel.send_outbound() → bus 记录
  → return ChannelMessage (HTTP Response)
```

**对接方式**: 只需提供符合 `(ChannelMessage, app_state) -> dict` 签名的 handler 函数，框架自动处理消息管道。

### 1.3 Agent 运行时（已有，部分复用）

```python
# runtime/agent_runner.py — 核心入口
run_agent(agent, user_content, tool_registry, skill_registry=None, 
          capability_selector=None, max_steps=3) -> str

run_agent_detailed(...) -> AgentRunResult  # 含 text, duration_ms, usage, model_calls
```

**对接方式**: 
- 论文精读 handler 调用 `run_agent_detailed()` 执行 agent
- 8 个 skill 通过 `SkillRegistry` + `CapabilitySelector` 动态加载
- 论文相关 tool 通过 `ToolRegistry` 注册，在 `profiles.json` 中声明

## 3. Phase 1: 基础设施 — 数据模型与存储

### 3.1 PaperReading 请求/响应协议

**设计原则**: 单个 `/paper_reading` POST 端点，通过 `action` 字段区分操作类型。这避免了对网关路由的大量修改。

#### 请求格式 (`paper_reading/schemas/request.py`)

```python
from pydantic import BaseModel, Field
from typing import Any, Literal
from uuid import UUID

class PaperReadingRequest(BaseModel):
    """论文精读统一请求体。"""
    action: Literal[
        "search_paper",       # 搜索论文
        "upload_paper",       # 上传 PDF（base64 编码）
        "start_reading",      # 开始/继续阅读
        "pause_reading",      # 暂停阅读
        "resume_reading",     # 恢复阅读
        "fork",               # 创建分支探索
        "merge",              # 合并分支
        "load_skill",         # 手动加载 skill
        "unload_skill",       # 手动卸载 skill
        "kg_query",           # KG 问答
        "get_session_state",  # 获取会话状态
        "get_progress",       # 获取阅读进度
    ]
    session_id: str | None = None       # 会话 ID（首次请求为空）
    paper_id: str | None = None         # 论文 ID
    content: str | None = None          # 用户消息/查询内容
    metadata: dict[str, Any] = {}       # 附加元数据
    # action 特定字段
    search_query: str | None = None     # search_paper 时的搜索词
    pdf_data: str | None = None         # upload_paper 时的 base64 PDF
    fork_target: str | None = None      # fork 时的目标内容（公式/段落引用）
    fork_skills: list[str] = []         # fork 时要加载的 skill
    fork_session_id: str | None = None  # merge 时的 fork 会话 ID
    skill_ids: list[str] = []           # load_skill/unload_skill 的目标 skill
    kg_question: str | None = None      # kg_query 时的问题
```

#### 存储实现 (`paper_reading/harness/storage.py`)

```python
import json
from pathlib import Path
from typing import Any

class PaperReadingStorage:
    """本地 JSON 文件持久化。"""
    
    def __init__(self, base_dir: Path | None = None):
        self.base_dir = base_dir or Path.home() / ".scholarsprout" / "paper_reading"
        for subdir in ["sessions", "papers", "kg", "uploads"]:
            (self.base_dir / subdir).mkdir(parents=True, exist_ok=True)
    
    def save_session(self, session_id: str, data: dict) -> None:
        path = self.base_dir / "sessions" / f"{session_id}.json"
        path.write_text(json.dumps(data, ensure_ascii=False, indent=2, default=str),
                        encoding="utf-8")
    
    def load_session(self, session_id: str) -> dict | None:
        path = self.base_dir / "sessions" / f"{session_id}.json"
        if not path.exists():
            return None
        return json.loads(path.read_text(encoding="utf-8"))
    
    def save_paper(self, paper_id: str, data: dict) -> None:
        path = self.base_dir / "papers" / f"{paper_id}.json"
        path.write_text(json.dumps(data, ensure_ascii=False, indent=2, default=str),
                        encoding="utf-8")
    
    def load_paper(self, paper_id: str) -> dict | None:
        path = self.base_dir / "papers" / f"{paper_id}.json"
        if not path.exists():
            return None
        return json.loads(path.read_text(encoding="utf-8"))
    
    def save_kg(self, paper_id: str, kg_data: dict) -> None:
        path = self.base_dir / "kg" / f"{paper_id}_kg.json"
        path.write_text(json.dumps(kg_data, ensure_ascii=False, indent=2, default=str),
                        encoding="utf-8")
    
    def load_kg(self, paper_id: str) -> dict | None:
        path = self.base_dir / "kg" / f"{paper_id}_kg.json"
        if not path.exists():
            return None
        return json.loads(path.read_text(encoding="utf-8"))
    
    def save_upload(self, paper_id: str, pdf_bytes: bytes) -> Path:
        path = self.base_dir / "uploads" / f"{paper_id}.pdf"
        path.write_bytes(pdf_bytes)
        return path
    
    def list_sessions(self, paper_id: str | None = None) -> list[dict]:
        sessions_dir = self.base_dir / "sessions"
        sessions = []
        for path in sessions_dir.glob("*.json"):
            data = json.loads(path.read_text(encoding="utf-8"))
            if paper_id is None or data.get("paper_id") == paper_id:
                sessions.append(data)
        return sessions
```

### 4.3 多源检索 (`paper_reading/pipeline/sources.py`)

```python
import httpx
import feedparser
from typing import Literal
from uuid import uuid4

class PaperSource:
    """多源论文检索基类，统一接口。"""
    
    SOURCE_NAME: str  # 子类覆盖
    
    async def search(self, query: str, max_results: int = 10) -> list[PaperMetadata]:
        raise NotImplementedError
    
    async def fetch_by_id(self, source_id: str) -> PaperMetadata:
        raise NotImplementedError

class ArxivSource(PaperSource):
    SOURCE_NAME = "arxiv"
    
    async def search(self, query: str, max_results: int = 10) -> list[PaperMetadata]:
        """通过 arXiv API 检索论文。
        
        API: http://export.arxiv.org/api/query?search_query=all:{query}&max_results={n}
        解析 Atom XML 响应，映射到 PaperMetadata。
        """
        url = f"http://export.arxiv.org/api/query?search_query=all:{query}&start=0&max_results={max_results}"
        async with httpx.AsyncClient() as client:
            response = await client.get(url, timeout=30.0)
            feed = feedparser.parse(response.text)
        
        papers = []
        for entry in feed.entries:
            arxiv_id = entry.id.split("/abs/")[-1]
            papers.append(PaperMetadata(
                paper_id=str(uuid4()),
                source="arxiv",
                source_id=f"arxiv:{arxiv_id}",
                title=entry.title.strip(),
                authors=[Author(name=author.name) for author in entry.authors],
                abstract=entry.summary.strip(),
                published_date=parse_date(entry.published),
                categories=[tag.term for tag in entry.tags],
                pdf_url=entry.id.replace("/abs/", "/pdf/"),
                url=entry.id,
            ))
        return papers

class SemanticScholarSource(PaperSource):
    SOURCE_NAME = "semantic_scholar"
    
    async def search(self, query: str, max_results: int = 10) -> list[PaperMetadata]:
        """通过 Semantic Scholar API 检索。
        
        API: https://api.semanticscholar.org/graph/v1/paper/search?query={query}&limit={n}
        响应已为 JSON，直接映射。
        """
        url = "https://api.semanticscholar.org/graph/v1/paper/search"
        params = {
            "query": query,
            "limit": max_results,

            "arxiv": ArxivSource(),
            "semantic_scholar": SemanticScholarSource(),
            # TODO: 后续添加 "dblp": DBLPSource(), "openalex": OpenAlexSource()
        }
        self.parser = PDFParser()
    
    async def search(self, query: str, sources: list[str] | None = None,
                     max_results: int = 10) -> list[PaperMetadata]:
        """多源并行搜索，合并去重。"""
        targets = sources or list(self.sources.keys())
        results = []
        for source_name in targets:
            if source_name in self.sources:
                try:
                    papers = await self.sources[source_name].search(query, max_results)
                    results.extend(papers)
                except Exception as e:
                    logger.warning(f"Source {source_name} search failed: {e}")
        return self._deduplicate(results)
    
    def parse_pdf(self, pdf_path: Path) -> PaperMetadata:
        """解析 PDF 文件，提取文本和元数据。"""
        return self.parser.parse(pdf_path)
    
    def _deduplicate(self, papers: list[PaperMetadata]) -> list[PaperMetadata]:
        """基于 DOI > arxiv_id > title 相似度去重。"""
        seen = set()
        unique = []
        for p in papers:
            key = p.doi or p.source_id or p.title.lower()[:50]
            if key not in seen:
                seen.add(key)
                unique.append(p)
        return unique

```python
import networkx as nx
from typing import Any

class KnowledgeGraphEngine:
    """论文知识图谱引擎。
    
    基于 networkx.DiGraph 存储，提供:
    - 节点/边 CRUD
    - 图遍历查询
    - 子图导出
    - 跨论文图融合
    """
    
    def __init__(self):
        self.graph = nx.DiGraph()
        self._paper_graphs: dict[str, nx.DiGraph] = {}  # paper_id → 子图缓存
    
    # ── 节点操作 ──
    
    def add_node(self, node: KGNode) -> str:
        """添加节点，返回 node_id。"""
        self.graph.add_node(
            node.node_id,
            node_type=node.node_type,
            label=node.label,
            paper_id=node.paper_id,
            section_id=node.section_id,
            properties=node.properties,
        )
        return node.node_id
    
    def get_node(self, node_id: str) -> dict | None:
        """获取节点数据。"""
        return dict(self.graph.nodes[node_id]) if node_id in self.graph else None
    
    def update_node(self, node_id: str, properties: dict) -> None:
        """更新节点属性。"""
        if node_id in self.graph:
            current = self.graph.nodes[node_id].get("properties", {})
            current.update(properties)
            self.graph.nodes[node_id]["properties"] = current
    
    # ── 边操作 ──
    
    def add_edge(self, source_id: str, target_id: str, edge: KGEdge) -> str:
        """添加边，返回 edge_id。"""
        self.graph.add_edge(
            source_id, target_id,
            edge_id=edge.edge_id,
            edge_type=edge.edge_type,
            paper_id=edge.paper_id,
            properties=edge.properties,
        )
        return edge.edge_id
    
    # ── 查询操作 ──
    
    def get_subgraph(self, paper_id: str) -> dict:
        """获取指定论文的 KG 子图（节点 + 边列表）。"""
        nodes = [
            {"id": n, **self.graph.nodes[n]}
            for n in self.graph.nodes
            if self.graph.nodes[n].get("paper_id") == paper_id
        ]
        edges = [
            {"source": u, "target": v, **self.graph.edges[u, v]}
            for u, v in self.graph.edges
            if self.graph.edges[u, v].get("paper_id") == paper_id
        ]
        return {"nodes": nodes, "edges": edges}
    
    def query_path(self, source_label: str, target_label: str) -> list[dict]:
        """查找两个实体间的因果/关联路径。
        
        用于 KG 驱动问答。示例:
        query_path("ProtoNet", "MAML") → 
          [outperforms edge + experiment + metric 信息]
        """
        results = []
        for u, v, data in self.graph.edges(data=True):
            u_label = self.graph.nodes[u].get("label", "")
            v_label = self.graph.nodes[v].get("label", "")
            if source_label in u_label and target_label in v_label:
                results.append({
                    "source": u_label,
                    "relation": data["edge_type"],
                    "target": v_label,
                    "properties": data["properties"],
                })
        return results
    
    def get_neighbors(self, node_id: str, depth: int = 1) -> dict:
        """获取节点的邻域子图。"""
        if depth == 1:
            predecessors = list(self.graph.predecessors(node_id))
            successors = list(self.graph.successors(node_id))
            return {
                "node": self.get_node(node_id),
                "incoming": [
                    {"from": p, "edge": self.graph.edges[p, node_id]}
                    for p in predecessors
                ],
                "outgoing": [
                    {"to": s, "edge": self.graph.edges[node_id, s]}
                    for s in successors
                ],
            }
        # depth > 1: 使用 nx.ego_graph
        subgraph = nx.ego_graph(self.graph, node_id, radius=depth)
        return {
            "nodes": [{"id": n, **self.graph.nodes[n]} for n in subgraph.nodes],
            "edges": [
                {"source": u, "target": v, **self.graph.edges[u, v]}
                for u, v in subgraph.edges
            ],
        }
    
    def to_cytoscape(self, paper_id: str) -> dict:
        """导出为 Cytoscape.js 兼容格式（前端可视化）。"""
        sg = self.get_subgraph(paper_id)
        elements = []
        for node in sg["nodes"]:
            elements.append({
                "data": {
                    "id": node["id"],
                    "label": node["label"],
                    "node_type": node["node_type"],
                    "properties": node["properties"],
                }
            })
        for edge in sg["edges"]:
            elements.append({
                "data": {
                    "id": edge["source"] + "_" + edge["target"],
                    "source": edge["source"],
                    "target": edge["target"],
                    "edge_type": edge["edge_type"],
                    "properties": edge["properties"],
                }
            })
        return {"elements": elements}

class KGQueryEngine:
    """基于 KG 结构化遍历的智能问答（非向量检索）。
    
    docx 示例:
    Q: "ProtoNet为什么比MAML好？"

    5. 遍历 Module → depends_on → Concept

        self.engine = engine
        self.model = model

        """KG 驱动问答入口。"""
        # 1. 提取问题中的实体
        entities = self._extract_entities(question)

                        "neighbors": neighbors,

        response = self.model.chat(messages=[{"role": "user", "content": prompt}])

            question=question,
            answer=response.choices[0].message.content,

## 触发时机
自动触发，当用户进入 Method 章节时激活。也可以在 Abstract 阅读后手动加载以获得方法概览。

## 核心行为
1. 拆解 Problem Formulation：输入/输出是什么？约束条件是什么？
2. 画出 Method Pipeline：列出方法的每个 Step
3. 标注每个 Step 的 Motivation：「为什么这个 Step 是必要的？」
4. 对核心创新 Step 进行深度分析
5. 输出结构化方法依赖图（DAG）

## 输出格式
严格输出以下 JSON：
```json
{
  "problem_formulation": {
    "input": "模型输入描述",
    "output": "模型输出描述",
    "constraints": ["约束1", "约束2"],
    "formal_definition": "正式的问题定义（如有公式，用 LaTeX）"
  },
  "pipeline": [
    {
      "step_id": "step_1",
      "name": "步骤名",
      "description": "做什么",
      "motivation": "为什么需要这一步",
      "is_core_innovation": false,
      "connects_to": ["step_2"]
    }
  ],
  "core_innovation_analysis": {
    "what": "核心创新是什么",
    "difference_from_prior": "与先前工作的差异",
    "why_it_works": "为什么这个创新有效（直觉解释）"
  },
  "dependency_graph": {
    "nodes": [{"id": "step_1", "label": "..."}],
    "edges": [{"from": "step_1", "to": "step_2", "label": "feeds_into"}]
  }
}
```

## 交互模式
苏格拉底式反问：不是直接告诉答案，而是引导思考「这个 Step 为什么必要？」

## 教学价值
学会「怎么看懂方法论」—— 这是科研阅读的核心能力。

用户主动加载，通常在读完 Experiment 后。

## 核心行为
以 Peer Review 标准逐条审查：
1. Assumption 是否合理？

6. 统计显著性是否报告？

### 7.1 会话管理器 (`paper_reading/harness/session.py`)

```python
from dataclasses import dataclass, field
from datetime import datetime, timezone
from uuid import uuid4
from typing import Literal

@dataclass
class Checkpoint:
    checkpoint_id: str = field(default_factory=lambda: str(uuid4()))
    position: dict = field(default_factory=dict)    # {section_id, paragraph_index}
    active_skills: list[str] = field(default_factory=list)
    kg_state_snapshot: dict = field(default_factory=dict)
    conversation_history: list[dict] = field(default_factory=list)
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())

@dataclass
class ReadingSession:
    session_id: str = field(default_factory=lambda: str(uuid4()))
    paper_id: str = ""
    user_id: str = ""
    state: Literal["active", "paused", "completed"] = "active"
    checkpoints: list[Checkpoint] = field(default_factory=list)
    progress: dict = field(default_factory=dict)      # {total_sections, completed, percentage}
    active_skills: list[str] = field(default_factory=list)
    created_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    updated_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
    # Fork 相关
    parent_session_id: str | None = None
    fork_sessions: list[str] = field(default_factory=list)  # 子 fork session ids
    fork_context: str | None = None                         # fork 时的上下文

class SessionManager:
    """会话生命周期管理。"""
    
    def __init__(self, storage: PaperReadingStorage):
        self.storage = storage
        self._active_sessions: dict[str, ReadingSession] = {}  # 内存缓存
    
    def create_session(self, paper_id: str, user_id: str = "default") -> ReadingSession:
        session = ReadingSession(paper_id=paper_id, user_id=user_id)
        self._active_sessions[session.session_id] = session
        self._persist(session)
        return session
    
    def get_session(self, session_id: str) -> ReadingSession | None:
        if session_id in self._active_sessions:
            return self._active_sessions[session_id]
        data = self.storage.load_session(session_id)
        if data:
            session = self._from_dict(data)
            self._active_sessions[session_id] = session
            return session
        return None
    
    def pause(self, session_id: str) -> ReadingSession:
        """暂停阅读，自动保存 checkpoint。"""
        session = self._active_sessions[session_id]
        # 创建 snapshot checkpoint
        checkpoint = Checkpoint(
            position=session.progress.get("current_position", {}),
            active_skills=session.active_skills.copy(),
            kg_state_snapshot={},   # 由 handler 填充
            conversation_history=[], # 由 handler 填充
        )
        session.checkpoints.append(checkpoint)
        session.state = "paused"
        session.updated_at = datetime.now(timezone.utc).isoformat()
        self._persist(session)
        return session
    
    def resume(self, session_id: str) -> ReadingSession:
        """恢复阅读，还原到最后 checkpoint。"""
        session = self.get_session(session_id)
        if session is None:
            raise ValueError(f"Session not found: {session_id}")
        session.state = "active"
        session.updated_at = datetime.now(timezone.utc).isoformat()
        self._active_sessions[session_id] = session
        self._persist(session)
        return session
    
    def update_progress(self, session_id: str, section_id: str, 
                        paragraph_index: int, completed: bool = False):
        session = self._active_sessions[session_id]
        session.progress["current_position"] = {
            "section_id": section_id,
            "paragraph_index": paragraph_index,
        }
        if completed:
            completed_sections = session.progress.get("completed_sections", [])
            if section_id not in completed_sections:
                completed_sections.append(section_id)
            session.progress["completed_sections"] = completed_sections
            total = session.progress.get("total_sections", 1)
            session.progress["percentage"] = len(completed_sections) / max(total, 1) * 100
        session.updated_at = datetime.now(timezone.utc).isoformat()
    
    def _persist(self, session: ReadingSession):
        self.storage.save_session(session.session_id, self._to_dict(session))
    
    def _to_dict(self, session: ReadingSession) -> dict:
        import dataclasses
        return dataclasses.asdict(session)
    
    def _from_dict(self, data: dict) -> ReadingSession:
        return ReadingSession(**data)
```

### 7.2 Fork/Merge 实现 (`paper_reading/harness/fork_merge.py`)

```python
class ForkMergeManager:
    """Fork/Merge 分支探索管理。"""
    
    def __init__(self, session_manager: SessionManager):
        self.session_manager = session_manager
    
    def create_fork(self, parent_session_id: str, fork_context: str,
                    fork_skills: list[str], user_id: str = "default") -> ReadingSession:
        """从主阅读流 fork 一个子会话。
        
        Args:
            parent_session_id: 主会话 ID
            fork_context: fork 时用户关注的内容（公式/段落/概念）
            fork_skills: fork 会话要加载的 skill
        """
        parent = self.session_manager.get_session(parent_session_id)
        if parent is None:
            raise ValueError(f"Parent session not found: {parent_session_id}")
        
        # 创建子会话（共享 paper_id）
        fork_session = self.session_manager.create_session(
            paper_id=parent.paper_id,
            user_id=user_id,
        )
        fork_session.parent_session_id = parent_session_id
        fork_session.fork_context = fork_context
        fork_session.active_skills = fork_skills
        
        # 更新父会话记录
        parent.fork_sessions.append(fork_session.session_id)
        self.session_manager._persist(parent)
        self.session_manager._persist(fork_session)
        
        return fork_session
    
    def merge_fork(self, fork_session_id: str) -> MergeResult:
        """将 fork 会话的成果合并回主阅读流。"""
        fork = self.session_manager.get_session(fork_session_id)
        if fork is None or fork.parent_session_id is None:
            raise ValueError(f"Invalid fork session: {fork_session_id}")
        
        parent = self.session_manager.get_session(fork.parent_session_id)
        if parent is None:
            raise ValueError(f"Parent session lost: {fork.parent_session_id}")
        
        # 收集 fork 中的 KG 增量
        # 收集 fork 中的关键发现（对话摘要）
        # 将这些合并到父会话
        
        return MergeResult(
            parent_session_id=parent.session_id,
            fork_session_id=fork.session_id,
            kg_updates=[],        # fork 中新增的 KG 节点
            key_findings=[],      # fork 中的关键结论
            merged_skills=fork.active_skills,
        )

```python
"""论文精读主 handler — 统一入口，按 action 路由到各子模块。"""

from __future__ import annotations

import asyncio
from typing import Any

from channels.base import ChannelMessage
from runtime.agent_runner import run_agent_detailed, AgentRunResult
from paper_reading.schemas.request import PaperReadingRequest
from paper_reading.schemas.response import PaperReadingResponse

async def handle_paper_reading_message(
    message: ChannelMessage,
    app_state: Any,
) -> dict[str, Any]:
    """论文精读统一 handler。
    
    对接框架接口: (ChannelMessage, app_state) -> dict
    """
    try:
        # 解析请求
        request = PaperReadingRequest.model_validate(
            message.content if isinstance(message.content, dict)
            else __import__("json").loads(message.content)
        )
    except Exception as e:
        return {"status": "error", "message": f"请求解析失败: {e}"}
    
    # 分发到对应 action 处理器
    handler_map = {
        "search_paper": _handle_search_paper,
        "upload_paper": _handle_upload_paper,
        "start_reading": _handle_start_reading,
        "pause_reading": _handle_pause_reading,
        "resume_reading": _handle_resume_reading,
        "fork": _handle_fork,
        "merge": _handle_merge,
        "load_skill": _handle_load_skill,
        "unload_skill": _handle_unload_skill,
        "kg_query": _handle_kg_query,
        "get_session_state": _handle_get_session_state,
        "get_progress": _handle_get_progress,
    }
    
    handler = handler_map.get(request.action)
    if handler is None:
        return {"status": "error", "message": f"未知 action: {request.action}"}
    
    return await handler(request, app_state)


async def _handle_start_reading(
    request: PaperReadingRequest, app_state: Any
) -> dict[str, Any]:
    """核心阅读逻辑。
    
    流程:
    1. 获取/创建 session
    2. 加载论文内容
    3. 构建 agent 消息（用户问题 + 论文上下文）
    4. 调用 run_agent_detailed() 执行
    5. 解析 agent 输出
    6. 触发 KG 构建
    7. 更新进度
    8. 返回结构化响应
    """
    session_manager = app_state.session_manager
    paper_pipeline = app_state.paper_pipeline
    kg_engine = app_state.kg_engine
    paper_agent = app_state.paper_reading_agent
    
    # 1. 会话管理
    session_id = request.session_id
    session = None
    if session_id:
        session = session_manager.get_session(session_id)
    
    if session is None:
        # 新会话
        session = session_manager.create_session(
            paper_id=request.paper_id or "unknown"
        )
    
    # 2. 加载论文
    paper = paper_pipeline.storage.load_paper(session.paper_id)
    if paper is None:
        return {"status": "error", "message": "论文未找到，请先上传或搜索"}
    
    # 3. 构建上下文
    current_section = session.progress.get("current_position", {}).get("section_id", "abstract")
    section_text = _get_section_text(paper, current_section)
    
    user_message = _build_agent_message(
        user_content=request.content or "",
        paper_context={
            "title": paper["title"],
            "section": current_section,
            "section_text": section_text,
            "kg_summary": kg_engine.get_subgraph(session.paper_id) if session.paper_id else {},
        },
        active_skills=session.active_skills,
    )
    
    # 4. 执行 agent
    result: AgentRunResult = run_agent_detailed(
        agent=paper_agent,
        user_content=user_message,
        tool_registry=app_state.tool_registry,
        skill_registry=app_state.skill_registry,
        capability_selector=app_state.capability_selector,
        max_steps=5,  # 论文精读可能需要更多 tool call 步骤
    )
    
    # 5. 解析 agent 输出
    parsed_output = _parse_agent_output(result.text, session.active_skills)
    
    # 6. 触发渐进式 KG 构建
    if app_state.kg_builder and session.paper_id:
        kg_update = app_state.kg_builder.build_for_section(
            paper_id=session.paper_id,
            paper_content=str(paper),
            section_id=current_section,
            section_text=section_text,
            existing_kg=kg_engine.get_subgraph(session.paper_id),
        )
    else:
        kg_update = None
    
    # 7. 更新进度
    session_manager.update_progress(
        session.session_id,
        section_id=current_section,
        paragraph_index=0  # 由前端精确追踪
    )
    
    # 8. 构建响应
    return {
        "status": "ok",
        "action": "start_reading",
        "data": {
            "session_id": session.session_id,
            "agent_response": result.text,
            "parsed_skill_outputs": parsed_output,
            "kg_update": kg_update.model_dump() if kg_update else None,
            "model_calls": result.model_calls,
            "duration_ms": result.duration_ms,
        },
        "session": {
            "session_id": session.session_id,
            "paper_id": session.paper_id,
            "state": session.state,
            "current_section": current_section,
            "active_skills": session.active_skills,
        },
        "progress": session.progress,

        # 实际实现中通过依赖注入传递
        engine = _get_kg_engine()
        query_type = arguments["query_type"]
        
        if query_type == "path":
            paths = engine.query_path(
                arguments.get("source_label", ""),

            )

            neighbors = engine.get_neighbors(

class PDFParseTool(BaseTool):
    def __init__(self):
        self.spec = ToolSpec(
            name="pdf_parse",
            description="解析已上传的 PDF 论文，提取指定章节内容。",
            parameters={
                "type": "object",
                "properties": {
                    "paper_id": {
                        "type": "string",
                        "description": "论文 ID",
                    },
                    "section_id": {
                        "type": "string",
                        "description": "要提取的章节 ID（如 'sec:3.2'），不指定则返回所有章节列表",
                    },
                },
                "required": ["paper_id"],
            },
        )
    
    def run(self, arguments: dict) -> dict:
        """提取论文章节内容。"""
        from paper_reading.harness.storage import PaperReadingStorage
        
        storage = PaperReadingStorage()
        paper = storage.load_paper(arguments["paper_id"])
        
        if paper is None:
            return {"error": f"论文未找到: {arguments['paper_id']}"}
        
        if "section_id" in arguments:
            section = next(
                (s for s in paper.get("sections", [])
                 if s.get("section_id") == arguments["section_id"]),
                None
            )
            if section:
                return {"section": section}
            return {"error": f"章节未找到: {arguments['section_id']}"}
        
        # 返回章节索引
        return {
            "sections": [
                {"section_id": s["section_id"], "title": s["title"], "level": s["level"]}
                for s in paper.get("sections", [])
            ]
        }
```
